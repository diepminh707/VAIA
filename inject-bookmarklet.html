// Manual Bridge Injection Script
// Paste this into the browser console on any page with Google AI Studio iframes

const injectExtensionBridge = () => {
    const bridgeScript = `
        const generateRequestId = () => {
            return 'req_' + Date.now() + '_' + Math.random().toString(36).substr(2, 9);
        };

        const serializeGoogleGenAI = (googleGenAI) => {
            try {
                const apiKey = googleGenAI.apiKey || extractApiKeyFromInstance(googleGenAI);
                if (!apiKey) {
                    throw new Error('Unable to extract API key from GoogleGenAI instance');
                }
                return JSON.stringify({
                    apiKey,
                    timestamp: Date.now()
                });
            } catch (error) {
                throw new Error('Failed to serialize GoogleGenAI instance: ' + (error.message || String(error)));
            }
        };

        const extractApiKeyFromInstance = (instance) => {
            if (instance.apiKey) return instance.apiKey;
            if (instance._apiKey) return instance._apiKey;
            if (instance.constructor && instance.constructor.apiKey) return instance.constructor.apiKey;

            let proto = Object.getPrototypeOf(instance);
            while (proto && proto !== Object.prototype) {
                if (proto.apiKey) return proto.apiKey;
                if (proto._apiKey) return proto._apiKey;
                proto = Object.getPrototypeOf(proto);
            }
            return null;
        };

        if (!window.__chromeExtensionBridge) {
            const pendingRequests = new Map();

            const extensionBridge = {
                call: (namespace, method, ...args) => {
                    return new Promise((resolve, reject) => {
                        const requestId = generateRequestId();
                        const timeout = setTimeout(() => {
                            pendingRequests.delete(requestId);
                            reject(new Error('Extension API call timeout'));
                        }, 5000);

                        pendingRequests.set(requestId, { resolve, reject });

                        window.postMessage({
                            type: 'extension_api_call',
                            namespace,
                            method,
                            args,
                            requestId,
                        }, '*');
                    });
                },

                execute: (googleGenAI, command) => {
                    return new Promise((resolve, reject) => {
                        const requestId = generateRequestId();
                        const timeout = setTimeout(() => {
                            pendingRequests.delete(requestId);
                            reject(new Error('GoogleGenAI execution timeout'));
                        }, 30000);

                        pendingRequests.set(requestId, { resolve, reject });
                        const serializedInstance = serializeGoogleGenAI(googleGenAI);

                        window.postMessage({
                            type: 'googlegenai_execute',
                            serializedInstance,
                            command,
                            requestId,
                        }, '*');
                    });
                },

                onResponse: (callback) => {
                    window.addEventListener('message', (event) => {
                        if (event.source !== window) return;
                        if (event.data.type === 'extension_api_response' || event.data.type === 'googlegenai_response') {
                            callback(event.data);
                        }
                    });
                },
            };

            window.addEventListener('message', (event) => {
                if (event.source !== window) return;
                if (event.data.type === 'extension_api_response' || event.data.type === 'googlegenai_response') {
                    const { requestId, result, error } = event.data;
                    const pending = pendingRequests.get(requestId);
                    if (pending) {
                        if (error) {
                            pending.reject(error);
                        } else {
                            pending.resolve(result);
                        }
                        pendingRequests.delete(requestId);
                    }
                }
            }, false);

            window.__chromeExtensionBridge = extensionBridge;
            console.log('[Manual Injection] Chrome Extension Bridge is now available');
        }
    `;

    // Inject into main window
    const mainScript = document.createElement('script');
    mainScript.textContent = bridgeScript;
    document.head.appendChild(mainScript);

    // Find and inject into Google AI Studio iframes
    const iframes = document.querySelectorAll('iframe');
    let injectedCount = 0;

    iframes.forEach((iframe, index) => {
        try {
            const src = iframe.src || '';
            if (src.includes('scf.usercontent.goog') ||
                src.includes('aistudio.google.com') ||
                iframe.title === 'Preview' ||
                (src.includes('google') && src.includes('ai'))) {

                const iframeDoc = iframe.contentDocument || iframe.contentWindow.document;

                if (iframeDoc && !iframe.contentWindow.__chromeExtensionBridge) {
                    const iframeScript = iframeDoc.createElement('script');
                    iframeScript.textContent = bridgeScript;

                    if (!iframeDoc.head) {
                        const head = iframeDoc.createElement('head');
                        iframeDoc.documentElement.appendChild(head);
                    }

                    iframeDoc.head.appendChild(iframeScript);
                    injectedCount++;
                    console.log(`✅ Injected bridge into iframe #${index} (src: ${src.substring(0, 50)}...)`);
                }
            }
        } catch (e) {
            console.log(`❌ Could not inject into iframe #${index}: ${e.message}`);
        }
    });

    console.log(`🚀 Manual injection completed. Processed ${iframes.length} iframes, successfully injected into ${injectedCount}`);
    return { totalIframes: iframes.length, injectedCount };
};

// Execute the injection
const result = injectExtensionBridge();
console.log('Injection result:', result);

✅ What This Does

  • Main Page: Injects window.__chromeExtensionBridge API
  • Google AI Studio iframes: Automatically detects and injects bridge into Google Cloud Functions iframes
  • Cross-origin safe: Handles access restrictions gracefully
  • Console logging: Reports injection success/failure for each iframe

⚠️ Important Notes

  • This manual injection only works if the iframe allows JavaScript execution
  • Cross-origin iframes may block access - this is a browser security feature
  • Your Chrome extension must be installed and loaded for the bridge to function
  • The bridge will only work if your extension's content script is also running on the page
